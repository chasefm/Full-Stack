Step 1建库建表

在 0002 课建好的 demo 库里执行（book 表如果已经建过就跳过，补一条即可）：

USE demo;

CREATE TABLE IF NOT EXISTS book (
    id         BIGINT AUTO_INCREMENT PRIMARY KEY,
    title      VARCHAR(100) NOT NULL COMMENT '书名',
    author     VARCHAR(50)  NOT NULL COMMENT '作者',
    status     TINYINT      NOT NULL DEFAULT 1 COMMENT '1上架 0下架',
    created_at DATETIME     DEFAULT CURRENT_TIMESTAMP
);
前端视角 这张表就是 TS 里的一个 interface Book——只是从「代码里声明形状」变成了「数据库里定义形状」。字段类型对应：BIGINT→Long、VARCHAR→String、TINYINT→Integer。表结构决定 Entity 的每个字段，两边不一致运行时就报错。

Step 3配置数据源

把 src/main/resources/application.properties 改名为 application.yml（公司项目几乎都用 yml），填入：

spring:
  datasource:
    driver-class-name: com.mysql.cj.jdbc.Driver
    url: jdbc:mysql://localhost:3306/demo?useSSL=false&serverTimezone=Asia/Shanghai&allowPublicKeyRetrieval=true
    username: root
    password: 你的MySQL密码
server:
  port: 8080

URL 里三个参数各有用途：useSSL=false 关掉本地不需要的加密握手；serverTimezone=Asia/Shanghai 防止时间字段差 8 小时；allowPublicKeyRetrieval=true 解决 MySQL 8 常见的连接报错。再加一个分页插件配置类——MyBatis-Plus 的分页必须靠它：

package com.demo.bookdemo.config;

import com.baomidou.mybatisplus.annotation.DbType;
import com.baomidou.mybatisplus.extension.plugins.MybatisPlusInterceptor;
import com.baomidou.mybatisplus.extension.plugins.inner.PaginationInnerInterceptor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

@Configuration                          // 配置类：往容器里注册 Bean
public class MybatisPlusConfig {

    @Bean
    public MybatisPlusInterceptor mybatisPlusInterceptor() {
        MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
        interceptor.addInnerInterceptor(new PaginationInnerInterceptor(DbType.MYSQL));
        return interceptor;
    }
}

Step 4四件套：Entity → Mapper → Service → Controller

在 com.demo.bookdemo 包下依次建 entity、mapper、service、controller 四个子包（对照 0001 的地图）。

① Entity：表的 Java 化身

package com.demo.bookdemo.entity;

import com.baomidou.mybatisplus.annotation.IdType;
import com.baomidou.mybatisplus.annotation.TableId;
import com.baomidou.mybatisplus.annotation.TableName;
import lombok.Data;

@Data                                              // Lombok：getter/setter/toString 全免
@TableName("book")                                 // 对应哪张表
public class Book {

    @TableId(type = IdType.AUTO)                   // 主键，自增
    private Long id;

    private String title;
    private String author;
    private Integer status;
}

对照你的 TS 心智：interface Book { id: number; title: string; author: string; status: 0 | 1 }——只是 Java 里它是一个真类，且字段名默认与表列名一一对应（驼峰自动映射下划线：createdAt ↔ created_at）。

③ Service：业务逻辑层

先建统一返回结构 common/Result.java——公司接口的标准姿势（code + message + data）：

package com.demo.bookdemo.common;

import lombok.Data;

@Data
public class Result<T> {
    private int code;        // 0 成功，非 0 失败
    private String message;
    private T data;

    public static <T> Result<T> ok(T data) {
        Result<T> r = new Result<>();
        r.code = 0;
        r.message = "success";
        r.data = data;
        return r;
    }

    public static <T> Result<T> error(String msg) {
        Result<T> r = new Result<>();
        r.code = 1;
        r.message = msg;
        return r;
    }
}

然后是 Service（直接单类，公司也有接口+Impl 风格，这里先用最简形式）：

package com.demo.bookdemo.service;

import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.demo.bookdemo.entity.Book;
import com.demo.bookdemo.mapper.BookMapper;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
@RequiredArgsConstructor                          // final 字段 → 构造器注入
public class BookService {

    private final BookMapper bookMapper;

    public List<Book> list(long pageNo, long pageSize) {
        Page<Book> page = bookMapper.selectPage(new Page<>(pageNo, pageSize), null);
        return page.getRecords();                 // records = 当前页数据；分页信息在 page 本身
    }

    public Book getById(Long id) {
        Book book = bookMapper.selectById(id);
        if (book == null) {
            throw new RuntimeException("图书不存在: " + id);   // 抛 unchecked，全局处理器接住
        }
        return book;
    }

    public void create(Book book) {
        bookMapper.insert(book);                  // 插入后 book.id 会被自动回填
    }

    public void update(Long id, Book book) {
        book.setId(id);
        bookMapper.updateById(book);              // 只更新非 null 字段
    }

    public void delete(Long id) {
        bookMapper.deleteById(id);
    }
}

④ Controller：接口层

package com.demo.bookdemo.controller;

import com.demo.bookdemo.common.Result;
import com.demo.bookdemo.entity.Book;
import com.demo.bookdemo.service.BookService;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/books")
@RequiredArgsConstructor
public class BookController {

    private final BookService bookService;

    @GetMapping                                   // GET /api/books?page=1&size=10
    public Result<List<Book>> list(@RequestParam(defaultValue = "1") long page,
                                   @RequestParam(defaultValue = "10") long size) {
        return Result.ok(bookService.list(page, size));
    }

    @GetMapping("/{id}")                          // GET /api/books/1
    public Result<Book> getById(@PathVariable Long id) {
        return Result.ok(bookService.getById(id));
    }

    @PostMapping                                  // POST /api/books
    public Result<Void> create(@RequestBody Book book) {   // @RequestBody：JSON 体 → Book 对象
        bookService.create(book);
        return Result.ok(null);
    }

    @PutMapping("/{id}")                          // PUT /api/books/1
    public Result<Void> update(@PathVariable Long id, @RequestBody Book book) {
        bookService.update(id, book);
        return Result.ok(null);
    }

    @DeleteMapping("/{id}")                       // DELETE /api/books/1
    public Result<Void> delete(@PathVariable Long id) {
        bookService.delete(id);
        return Result.ok(null);
    }
}

Step 6React 页面连上接口

这步是你的主场，重点是对准接口形状：注意后端返回被 Result 包了一层，前端要解两次 .data（axios 的 data → Result 的 data）。新开一个 Vite React 项目（或直接在你任意练手前端里），替换 App.jsx：

import { useEffect, useState } from "react";

const BASE = "http://localhost:8080/api/books";

export default function App() {
  const [books, setBooks] = useState([]);
  const [form, setForm] = useState({ title: "", author: "" });

  const load = async () => {
    const res = await fetch(`${BASE}?page=1&size=10`);
    const json = await res.json();        // json = { code, message, data }
    setBooks(json.data);                  // Result 解包：data 里才是列表
  };

  useEffect(() => { load(); }, []);

  const add = async (e) => {
    e.preventDefault();
    await fetch(BASE, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ ...form, status: 1 }),
    });
    setForm({ title: "", author: "" });
    load();
  };

  const toggle = async (b) => {
    await fetch(`${BASE}/${b.id}`, {
      method: "PUT",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ status: b.status === 1 ? 0 : 1 }),
    });
    load();
  };

  return (
    <div style={{ maxWidth: 560, margin: "40px auto", fontFamily: "sans-serif" }}>
      <h2>📚 图书管理</h2>
      <form onSubmit={add} style={{ display: "flex", gap: 8, marginBottom: 16 }}>
        <input value={form.title} required placeholder="书名"
               onChange={(e) => setForm({ ...form, title: e.target.value })} />
        <input value={form.author} required placeholder="作者"
               onChange={(e) => setForm({ ...form, author: e.target.value })} />
        <button>新增</button>
      </form>
      <ul>
        {books.map((b) => (
          <li key={b.id} style={{ display: "flex", gap: 8, alignItems: "center" }}>
            <span style={{ flex: 1 }}>
              {b.title} · {b.author} · {b.status === 1 ? "上架" : "下架"}
            </span>
            <button onClick={() => toggle(b)}>{b.status === 1 ? "下架" : "上架"}</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
跨域注意 页面跑在 5173、接口跑在 8080，浏览器会拦截跨域请求。两个解法任选：Vite 里配 server.proxy 把 /api 转发到 8080（前端正规军做法）；或后端加一个 @CrossOrigin 注解在 Controller 上（后端解法，最快见效）。

练习场

练习 1 · bookMapper.selectById(1) 最终执行的 SQL 是？

解析：BaseMapper 按方法名 + Entity 注解自动生成 SQL：selectById → 按主键查全字段。这就是 MyBatis-Plus「白送 CRUD」的含义——你没有写一行 SQL，它替你写了。

练习 2 · 前端 fetch 拿到响应后，书单列表在哪个字段里？

解析：两层解包——HTTP 响应体先 json() 一次得到 Result 结构 {code, message, data}，列表在 data 里。以后你会看到公司前端全是 res.data.data 这种双解包，源头就是统一返回结构 Result<T>。

练习 3 · Service 里 throw new RuntimeException("图书不存在") 之后，谁接住了它？

解析：unchecked 异常沿调用栈向上冒泡，Spring 的异常处理机制统一接住转成 500 响应。真实项目里会有 @RestControllerAdvice 定制返回格式——这正是 0003 课讲的「公司代码很少 try/catch」的答案。

凭回忆作答（不看上文）

检索练习是记忆真正的来源——先在心里答完，再点开核对。

Q1：四件套各自的职责一句话概括？

Entity 定数据形状（对应表）；Mapper 白拿 CRUD（BaseMapper）；Service 干业务活；Controller 接 HTTP 请求并统一 Result 返回。

Q2：Spring Boot 3 配 MyBatis-Plus，starter 依赖叫什么名字？

com.baomidou:mybatis-plus-spring-boot3-starter（带 spring-boot3 后缀；不带的是给 Boot 2 用的）。

Q3：接口统一返回 Result<T>（code/message/data）对前端有什么好处？

前端可以写统一的响应拦截器：code 非 0 就报错提示，为 0 就解包 data——不用每个接口各自处理错误形状。
